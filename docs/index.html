Supabase 数据库配置

免费、永久、多设备同步。数据存在云端 PostgreSQL。

保存后，任何设备打开网站时无需重新配置。Token 需要 Contents: Read and write 权限。 申请 Token →

数据库初始化 SQL

在 Supabase → SQL Editor 里执行以下 SQL（首次配置时运行一次）：

-- 运动记录表
CREATE TABLE workout_logs (
  id         BIGSERIAL PRIMARY KEY,
  date       DATE NOT NULL,
  exercises  JSONB DEFAULT '[]',
  duration   INTEGER DEFAULT 0,
  notes      TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 营养记录表
CREATE TABLE nutrition_logs (
  id         BIGSERIAL PRIMARY KEY,
  date       DATE NOT NULL,
  calories   INTEGER DEFAULT 0,
  protein    DECIMAL(6,1) DEFAULT 0,
  carbs      DECIMAL(6,1) DEFAULT 0,
  fat        DECIMAL(6,1) DEFAULT 0,
  notes      TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 体测数据表
CREATE TABLE body_metrics (
  id          BIGSERIAL PRIMARY KEY,
  measured_at TIMESTAMPTZ NOT NULL,
  weight      DECIMAL(5,2),
  body_fat    DECIMAL(5,2),
  waist       DECIMAL(5,1),
  hip         DECIMAL(5,1),
  notes       TEXT DEFAULT '',
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
-- 已有表升级（如已建表则执行这两行）
-- ALTER TABLE body_metrics ADD COLUMN IF NOT EXISTS waist DECIMAL(5,1);
-- ALTER TABLE body_metrics ADD COLUMN IF NOT EXISTS hip   DECIMAL(5,1);

-- 训练计划（主表）
CREATE TABLE training_plans (
  id          BIGSERIAL PRIMARY KEY,
  name        TEXT NOT NULL,
  description TEXT NOT NULL DEFAULT '',
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 热身项目
CREATE TABLE plan_warmup (
  id         BIGSERIAL PRIMARY KEY,
  plan_id    BIGINT NOT NULL REFERENCES training_plans(id) ON DELETE CASCADE,
  sort_order SMALLINT NOT NULL DEFAULT 0,
  name       TEXT NOT NULL,
  duration   TEXT NOT NULL DEFAULT '',
  speed      TEXT NOT NULL DEFAULT '',
  note       TEXT NOT NULL DEFAULT ''
);

-- 训练动作
CREATE TABLE plan_exercises (
  id          BIGSERIAL PRIMARY KEY,
  plan_id     BIGINT NOT NULL REFERENCES training_plans(id) ON DELETE CASCADE,
  sort_order  SMALLINT NOT NULL DEFAULT 0,
  name        TEXT NOT NULL,
  category    TEXT NOT NULL DEFAULT '',
  tag         TEXT NOT NULL DEFAULT '',
  rest_time   TEXT NOT NULL DEFAULT '',
  target      TEXT NOT NULL DEFAULT '',
  warning     TEXT NOT NULL DEFAULT '',
  notes       TEXT[] NOT NULL DEFAULT '{}',
  warmup_sets TEXT[] NOT NULL DEFAULT '{}',
  conditions  JSONB NOT NULL DEFAULT '[]'
);

-- 目标组数据
CREATE TABLE plan_sets (
  id          BIGSERIAL PRIMARY KEY,
  exercise_id BIGINT NOT NULL REFERENCES plan_exercises(id) ON DELETE CASCADE,
  set_number  SMALLINT NOT NULL DEFAULT 1,
  reps        TEXT NOT NULL DEFAULT '',
  weight_lb   DECIMAL(6,2),
  is_warmup   BOOLEAN NOT NULL DEFAULT FALSE
);

-- 食物库
CREATE TABLE food_library (
  id         BIGSERIAL PRIMARY KEY,
  name       TEXT NOT NULL,
  protein    DECIMAL(6,2) NOT NULL DEFAULT 0,
  carbs      DECIMAL(6,2) NOT NULL DEFAULT 0,
  fat        DECIMAL(6,2) NOT NULL DEFAULT 0,
  calories   INTEGER,
  unit_name  TEXT NOT NULL DEFAULT '',
  unit_grams DECIMAL(6,1),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 给营养记录表添加食物明细列（已有数据库执行此行即可）
ALTER TABLE nutrition_logs ADD COLUMN IF NOT EXISTS food_items JSONB DEFAULT '[]';

-- 开放读写权限（个人使用）
ALTER TABLE workout_logs    ENABLE ROW LEVEL SECURITY;
ALTER TABLE nutrition_logs  ENABLE ROW LEVEL SECURITY;
ALTER TABLE body_metrics    ENABLE ROW LEVEL SECURITY;
ALTER TABLE training_plans  ENABLE ROW LEVEL SECURITY;
ALTER TABLE plan_warmup     ENABLE ROW LEVEL SECURITY;
ALTER TABLE plan_exercises  ENABLE ROW LEVEL SECURITY;
ALTER TABLE plan_sets       ENABLE ROW LEVEL SECURITY;
ALTER TABLE food_library    ENABLE ROW LEVEL SECURITY;

CREATE POLICY "open" ON workout_logs   FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON nutrition_logs FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON body_metrics   FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON training_plans FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON plan_warmup    FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON plan_exercises FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON plan_sets      FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON food_library   FOR ALL TO anon USING (true) WITH CHECK (true);
