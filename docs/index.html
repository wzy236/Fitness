Supabase 数据库配置

免费、永久、多设备同步。数据存在云端 PostgreSQL。

数据库初始化 SQL

在 Supabase → SQL Editor 里执行以下 SQL（首次配置时运行一次）：

-- 运动记录表
CREATE TABLE workout_logs (
  id         BIGSERIAL PRIMARY KEY,
  date       DATE NOT NULL,
  exercises  JSONB DEFAULT '[]',
  duration   INTEGER DEFAULT 0,
  notes      TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 营养记录表
CREATE TABLE nutrition_logs (
  id         BIGSERIAL PRIMARY KEY,
  date       DATE NOT NULL,
  calories   INTEGER DEFAULT 0,
  protein    DECIMAL(6,1) DEFAULT 0,
  carbs      DECIMAL(6,1) DEFAULT 0,
  fat        DECIMAL(6,1) DEFAULT 0,
  notes      TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 体测数据表
CREATE TABLE body_metrics (
  id          BIGSERIAL PRIMARY KEY,
  measured_at TIMESTAMPTZ NOT NULL,
  weight      DECIMAL(5,2),
  body_fat    DECIMAL(5,2),
  notes       TEXT DEFAULT '',
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- 开放读写权限（个人使用）
ALTER TABLE workout_logs  ENABLE ROW LEVEL SECURITY;
ALTER TABLE nutrition_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE body_metrics  ENABLE ROW LEVEL SECURITY;

CREATE POLICY "open" ON workout_logs  FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON nutrition_logs FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "open" ON body_metrics  FOR ALL TO anon USING (true) WITH CHECK (true);
  1. 注册 supabase.com，新建项目
  2. 进入项目 → SQL Editor → 粘贴上方 SQL → Run
  3. 进入 Project Settings → API，复制 Project URL 和 anon key
  4. 粘贴到上方输入框 → 保存 → 测试连接
